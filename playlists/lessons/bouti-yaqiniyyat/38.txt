من ذلك الحديث المتفق عليه المعروف الذي يرويه عمر بن الخطاب رضي الله عنه قال بينما نحن جلوس عند رسول الله صلى الله عليه وسلم إذ دخل علينا رجل شديد بياض الثياب شديد سواد الشعر لا يرى عليه أثر السفر ولا يعرفه منا أحد، الحديث طويل، في آخر الحديث سأل هذا الرجل رسول الله صلى الله عليه وسلم عن الساعة، يعني متى الساعة؟
قال: ما المسؤول عنها بأعلم من السائل،

أول هذه الأحاديث ما رواه الشيخان وغيرهما من حديث عبد الله بن عمر رضي الله عنهما أن رسول الله صلى الله عليه وسلم قام فخطب الناس فأثنى على الله عز وجل بما هو أهله ثم ذكر الدجال فقال: إني لأنذركموه، يعني أنا أحذركم منه، وما من نبي إلا وقد أنذر قومه، ولكني سأقول لكم قولا لم يقله نبي قبلي، إنه أعور وإن ربكم ليس بأعور، نعم، هذا حديث من الأحاديث ذكرها رسول الله صلى الله عليه وسلم.
حديث آخر أيضا يرويه الشيخان، بخاري ومسلم وغيرهما من حديث حذيفة رضي الله عنه، واللفظ التالي لمسلم، يعني هو متفق عليه لكن اللفظ
هذا الذي نذكره لمسلم في صحيحه: أن عقبة رضي الله عنه سأل حذيفة: ما سمعت من رسول الله صلى الله عليه وسلم عن الدجال؟ فقال: صلى الله عليه وسلم: إن الدجال يخرج وإن معه ماء ونارا، يعني معه نعيم يعبر عنه بالماء، ومعه تعذيب يعبر عنه بالنار، فاما الذي يراه الناس ماء فنار تحرق، وأما الذي يراه الناس نارا فماء بارد عذب، هكذا يقول رسول الله، فمن أدرك ذلك منكم فليقع في الذي يراه نارا فإنه عذب طيب، فقال عقبة: وأنا سمعت ذلك من رسول الله صلى الله عليه وعلى آله وسلم.
أيضا حديث آخر وهذا الحديث أكثر تفصيلا يقول فيه رسول الله صلى الله عليه وسلم، والحديث يرويه مسلم في صحيحه وأبو داود والترمذي وابن ماجه وأحمد، نعم، وآخرون، نعم، حديث طويل ذكره رسول الله صلى الله عليه وسلم عن الدجال، نعم، يرويه النواس بن سمعان رضي الله عنه قال: ذكر رسول الله صلى الله عليه وسلم الدجال ذات غداة فخفض فيه ورفع، يعني ذكر من خطورته ما ذكر، حتى ظنناه في طائفة النخل، يعني قريبا منا يمكن عما قريب سيظهر، فلما رحنا إليه عرف ذلك فينا، يعني عرف تخوفنا من ظهوره، نعم، بين ظهرانينا يعني، فقال عليه الصلاة والسلام: غير الدجال أخوفني عليكم، يعني أنا لا أخاف عليكم إلا من العوامل التي تجعلكم ضحية للدجال.
كل إنسان بقي مستقيما على دين الله ملتزما لأوامره يراقب الله عز وجل ويصلح أمره كلما زلت به القدم، لا خوف عليه من الدجال، لكن الفتن التي ستأتي قبل ظهوره والتي تجعل الناس يشردون عن دين الله عز وجل ويقعون في المحرمات والبدع وما إلى ذلك هذا هو الذي أخافه عليكم، هذا معنى كلام رسول الله صلى الله عليه وسلم، إن يخرج وأنا فيكم فأنا حجيجه، لا تخافوا يعني، وإن يخرج ولست فيكم فامرؤ حجيجه، كل واحد يعني يدير باله على حاله ضد، والله خليفتي على كل مسلم، إنه شاب قطط، لا، اسمعوا أيها الإخوة، هل يتأتى لنا أن نؤول الدجال بجهاز؟
هل يتأتى لنا أن نؤول الدجال بشيء مما يذكره هؤلاء المتلاعبون بدين الله كالذين يتلاعبون بكرة القدم؟ النبي عليه الصلاة والسلام يصفه يقول: إنه شاب قطط يعني شعره أجعد أجعد، نعم، بشدة، عينه طافية، العين الواحدة منهن يعني جاحظة خارجة، نعم، كأني أشبهه بعبد العزى بن قطن، لك عم يشبهه برجل، فمن أدركه منكم فليقرأ عليه فواتح سورة الكهف، إنه خارج خلة فجأة بين الشام والعراق، يعني يخرج
ما بين الشام والعراق من جهة الشرق، فعاث يمينا وعاث يسارا، يا عباد الله فاثبتوا، يعني لا يخدعنكم بدجله،
قلنا: يا رسول الله، نعم، وما لبثه في الأرض؟
قال: أربعون يوما، يوم كسنة، ويوم كشهر، ويوم كجمعة، وسائر أيامه كأيامكم،
قلنا: يا رسول الله، فذلك اليوم الذي كسنة أتكفينا فيه صلاة يوم؟ قال: اقدروا له قدره.
قلنا: يا رسول الله وما إسراعه في الأرض؟
قال: كالغيث استدبرته الريح، فيأتي على القوم فيدعوهم فيؤمنون به ويستجيبون له، يأمر السماء فتمطر والأرض فتنبت، ثم يدعو رجلا ممتلئا شبابا فيضربه بالسيف، يعني يضربه بأي أداة ربما كان سيف نعم، فيقطعه جزلتين، قطعتين، رمية الغرض، ثم يدعوه فيقبل ويتهلل وجهه يضحك، فبينما هو كذلك إذ بعث الله المسيح ابن مريم فينزل عند المنارة البيضاء شرقي دمشق، معروفة نحن نراها اليوم، منارة ما فيها مسجد بس منارة، نعم، بين مهرودتين يعني يلبس ثوبين أصفرين، واضعا كفيه على أجنحة ملكين، إذا طأطأ رأسه قطر ماء يعني، وإذا رفعه تحدر منه مثل الجمان، فيطلبه، يعني يطلب الدجال حتى يدركه بباب لد فيقتله، طيب.
هذه طائفة من الأحاديث التي يذكرها المصطفى صلى الله عليه وسلم عن الدجال.
المهم من هذه الأحاديث التي ذكرتها وهي طائفة يسيرة جدا من الأحاديث التي ذكرها لنا رسول الله صلى الله عليه وسلم المقصود من ذلك أن نعلم أنه رجل من الناس، ولا يجوز لإنسان أبدا أن يتلاعب بكلام المصطفى صلى الله عليه وسلم فيؤول هذا الكلام الذي لا يقبل التأويل، هو يصف طوله وعرضه، شاب شعره أجعد عين واحدة وإلى آخره ما هنالك، ويظهر في أي مكان، يسير، يتكلم، نعم.
الشيء الثاني الذي ينبغي أن نعلمه، وأن نؤمن بأن من أشراط الساعة ومن أوائل أشراط الساعة الكبرى ظهور الدجال.
الشيء الأهم من هذا كله أن نعلم أن الإنسان لا يمكن أن يتقي شر الدجال ونسأل الله أن لا يبلغنا يومه، لكن لو أن في الناس من قد يدركونه ليست وسيلة الحماية منه متمثلة في قوة مادية، قوته أقوى من كل قوة آنذاك، وإنما وسيلة التحصن ضده كما أشار المصطفى صلى الله عليه وسلم كثرة التجائه إلى الله، وشدة اعتصامه بكتاب الله سبحانه وتعالى، وبعده عن الانحراف ما وسعه الجهد، فإن زلت به القدم كما قلنا الساعة فالتوبة فالتوبة إلى الله سبحانه وتعالى هو المغتسل، يعني إذا كان الماء الذي أنزله الله عز وجل هو أداة الطهارة من الأرجاس المادية، فالتوبة هي الماء
الذي أنزله الله عز وجل للعاصين ليتخلصوا بهذه التوبة من أوزار معاصيهم، نعم، التوبة باب التوبة مفتوح، فإذا كان الإنسان معتصما بكتاب الله، ملتزما بهدي رسول الله، يحاذر من الانحراف جهد الاستطاعة فإن انحرف آب وتاب إلى الله سبحانه وتعالى هذا هو الذي يعصمه من الدجال، وليس الذي يعصمه تحايل هو دجله يعني يتغلب على دجل كل من في الأرض، كل من على وجه الأرض، نعم، وحسبكم دليلا على خطورته أن المصطفى صلى الله عليه وسلم ندبنا إلى أن ندعو في آخر كل صلاة أن يقينا شره: اللهم إني أعوذ بك من عذاب القبر ومن عذاب النار ومن فتنة المسيح الدجال.
هذا من دعاء رسول الله صلى الله عليه وعلى آله وسلم عندما ينتهي من التشهد وقبل أن يسلم، لاحظوا الأمرين الذين درسناهما وكلاهما غيبيين: اللهم إني أعوذ بك من عذاب القبر، ومن عذاب النار يوم القيامة، ومن فتنة المحيا والممات فتن كثيرة، ومن فتنة المسيح الدجال، طيب.
هذا إذاً واحد أيضا من الأمور الغيبية التي لا يجوز للإنسان أن يعني ينكرها.
بقيت هنالك ربما بعض الإشكالات، قد يقول قائل: طيب إنه النبي وصفه بصفات لا تنطبق إلا على ذات الله، يقطع زيدا من الناس قطعتين ثم إنه يعيدهما مرة ثانية إلى الحياة، أليس هذا دليلا على أن هذا ليس رجلا؟ أليس هذا دليلا على أنه يعني ربما حسب ما يتخيل البعض إله؟ ما دام هيك عنده قدرة يحيي الموتى، النمرود شو قال لسيدنا إبراهيم؟ قال: أنا أحيي وأميت، نعم، ألم تر إلى الذي حاج إبراهيم في ربه أن آتاه الله الملك إذ قال إبراهيم ربي الذي يحيي ويميت، قال إبراهيم أنا أحيي، قال أنا أحيي وأميت النمرود، طيب، سيدنا إبراهيم حصر الإحياء والموت في ذات الله، قال إبراهيم ربي الذي يحيي ويميت، ها نحن هنا نجد إنه الدجال كذلك يحيي ويميت، كيف هذا؟
أولا الدجال شخص فريد من نوعه وقد شاء الله أن يجعله فتنة لأهل عصره، فهو مستثنى عن هذا القانون الذي ذكره سيدنا إبراهيم للنمرود، طيب، أما إن كانت هذه الظاهرة تجعل عند الناظر إليه حالة التباس يعني تجعله يعني يظن أنه إله إذاً، عم يحيي ويميت وما إلى ذلك أخص على يعني صفات الربوبية أنه يحيي الموتى، نعم، فما الجواب؟
الجواب كما أوضح المصطفى صلى الله عليه وسلم في بعض الأحاديث إذا ساورتك نفسك وتسربت الشبهات إلى عقلك فتذكر إنه هذا الإله لازم قبل كل شيء شو يساوي؟
يزيل هاللتشوه القذر على وجهه، واضح كيف؟ ما دمت أنت إله قبل ما تحيي وتموت وتقول أنا كذا، صلح خلقتك حتى فعلا تصير إله، لكنه لا يقوى على ذلك.
إذاً ففي الوقت الذي نعم أمكنه الله عز وجل لحكمة من أن يفعل هذه الأمور، ولكن الله عز وجل جعل أمامك وبين يديك حجة قاصمة تدرك بها أن الرجل دجال، وأنه ليس بإله، هو ادعى الألوهية، عم تدعي الألوهية وشكلك مشوه من فوق لتحت، نعم.
إذاً هذا هو الجواب عن هذا الإشكال.
ثم ورد أيضا أنه هذا الإنسان الذي يقطعه قطعتين ثم يعيده إلى الحياة، يقول له: الآن تأكدت أنك أنت الدجال، لأن النبي عليه الصلاة والسلام وصفك لي، أنه بيعمل هالعمل، هلا أنت عملت مثل ما قال النبي عليه الصلاة والسلام، فأن الآن لم يبق لدي ولا احتمال في المليون أنك لست، أنت هو الدجال ذاته، نعم.
والمهم كما قال المصطفى صلى الله عليه وسلم: غير الدجال أخوفني عليكم، أنا ما لي خايف عليكم من الدجال، خايف عليكم من المقدمات التي تجعلكم ضحية للدجال.
ما هي هذه المقدمات؟
مقدمات ابتعادنا عن الدين، تقليدنا لأعداء الله سبحانه وتعالى، وينبغي أن نعلم أن بين يدي الدجال الكبير دجاجلة، هؤلاء الدجاجلة يمهدون له، والدنيا الآن مليئة بهؤلاء الدجاجلة.
عندما يسيل لعابنا على اتباع أولئك الأوغاد، أولئك الدجاجلة، وعندما نسير سيرتهم، وننسى ديننا الذي أعزنا الله عز وجل به، ربي خلقنا لمهمة، المهمة طويناها وألقيناها وراءنا ظهريا، واخترعنا عوامل ووظائف أخرى، الحداثة، العلمانية، إلى آخره ما هنالك، نعم، وكل ما جاءنا شيء من الغرب، ولو عن طريق أيا كان، أو مغطى باسم هيئة الأمم المتحدة، هيئة الأمم المتحدة عم تتكلم باسم الشعوب يا أخي؟ هيئة الأمم المتحدة عم تتكلم باسم العالم؟ باسم المجتمع؟ لا، هيئة الأمم المتحدة في الواقع الآن تتكلم وتستنطق باسم أعتى طغيان الأرض، أعتى طغاة الأرض.
لما بنجي بناخد قوانينهم اللي جاية من هيئة الأمم المتحدة، يلا، بتكون طبقوها، من طبقها، يا أخي تحفظوا عليها، تستطيعون أن تتحفظوا عليها وتمزقوها، لا لا لا، ما بدنا نتحفظ عليها، وبدنا نجي نحطها على رأسنا، ونقدسها ونقدرها، هذا التصرف إذا استمر هو الذي يجعل الناس ضحايا للدجال، الدجال انعكاس لهذا الواقع، نعم.
هذه حقيقة ينبغي أن نعلمها.
الآن ننتقل إلى أمر آخر من الأمور الغيبية، لكن سنشرحه، وهو نزول سيدنا عيسى عليه الصلاة والسلام في درس قادم، والحمد لله رب العالمين.
اللهم لك الحمد وإليك المشتكى وأنت المستعان وعليك التكلان.
اللهم صل أفضل صلواتك على أسعد مخلوقاتك سيدنا محمد النبي الأمي وعلى آله وصحبه وسلم تسليما.
اللهم يا من لا يشغله سمع عن سمع، ويا من لا تغلطه كثرة المسائل، ويا من لا يتبرم من إلحاح عباده الملحين عليه، أذقنا برد رحمتك،
وكرم استجابتك يا رب العالمين.
اشرح اللهم صدورنا، أذهب اللهم غم نفوسنا، اجمع اللهم لنا الخير كله، واصرفنا عن الشر كله، يا رب اجعلنا واجعل إخواننا جميعا في حمايتك وفي حرزك وفي كنفك من شر أنفسنا ومن شر خلقك ومن شر كل سوء يا رب العالمين.
اجعل اللهم هذه البلدة بلدة آمنة مطمئنة رخية مستظلة بظل كتابك ملتزمة بهدي نبيك محمد صلى الله عليه وعلى آله وسلم وسائر بلاد المسلمين يا رب العالمين.
يا رب من أراد بالإسلام والمسلمين خيرا فوفقه اللهم إلى كل خير، ومن أراد بالإسلام والمسلمين شرا فخذه اللهم أخذ عزيز مقتدر، يا قيوم السماوات والأرض، يا ذا الجلال والإكرام، يا ذا الطول والإنعام، عزمنا عليك بذل عبوديتنا لك، وبعظيم افتقارنا إليك إلا استجبت دعاءنا وحققت رجاءنا، أعط كل واحد منا في هذه الساعة سؤله يا أرحم من سئل ويا أكرم من أعطى، يا ذا الجلال والإكرام، يا ذا الجلال والإكرام، يا ذا الجلال والإكرام، يا أرحم الراحمين، يا أرحم الراحمين، يا أرحم الراحمين، استودعنا لديك دعاءنا ورجاءنا وآمالنا، فاستجب اللهم دعاءنا وحقق رجاءنا ولا تخيبنا في آمالنا، وصلى الله على سيدنا محمد النبي الأمي وعلى آله وصحبه أجمعين، والحمد لله رب العالمين.
